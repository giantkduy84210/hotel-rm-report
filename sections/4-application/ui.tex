\subsection{Streamlit User Interface}
\begin{guide}
  \guidepart{Content to include}
  \begin{itemize}
    \item 4 screens: Dashboard (KPIs + OLAP), Forecasting, Optimisation \& What-if (all M1--M5 parameters as interactive controls, assumptions displayed alongside), Scenario Comparison (reads \texttt{fact\_scenario\_run} back)

    \item Screenshots of each screen
    \item User flow following Simon's 4 phases
  \end{itemize}
